\section{Khai báo sử dụng AI}
\label{sec:sources}
% ============================================================

Quy định của môn yêu cầu khai báo việc dùng AI trong báo cáo \cite{spec}; \cref{tab:ai} liệt kê
phạm vi hỗ trợ. \textbf{Khi chạy, hệ thống không dùng LLM hay mô hình học máy nào}; AI chỉ
hỗ trợ trong quá trình phát triển.

\begin{table}[htbp]
  \centering
  \small
  \begin{tabularx}{\linewidth}{|P{3cm}|L|}
    \hline
    \textbf{Công cụ} & \textbf{Phạm vi hỗ trợ} \\ \hline
    Claude Code: Claude Opus, Claude Sonnet; 22/10--19/11/2026 &
      Gộp grammar giai đoạn 1; hợp đồng giữa các tầng và công cụ sinh golden test;
      pipeline, CLI, xuất output, đánh giá; sửa tokenizer và bộ sinh câu; kiểm chứng
      Docker; soạn và biên tập báo cáo \LaTeX. Commit liên quan có dòng
      \texttt{Co-Authored-By: Claude}. \\ \hline
    Codex; 16--17/11/2026 &
      Sửa luật rỗng và kiểm tra mơ hồ của parser; mở rộng grammar và semantic cho tính
      năng BTL; câu thử thách C009--C022; công cụ kiểm tra grammar và đóng gói; nháp
      README, báo cáo, kịch bản demo; bổ sung test phản ví dụ. \\ \hline
  \end{tabularx}
  \caption{Khai báo sử dụng công cụ AI}
  \label{tab:ai}
\end{table}

Mọi thay đổi có AI hỗ trợ đều phải qua test hợp đồng, test hồi quy và parse lại toàn bộ câu
sinh, rồi được thành viên review trước khi đưa vào bản nộp. Nhóm không sửa KB để chữa ca
sai và chịu trách nhiệm giải thích được toàn bộ mã nộp. Ngoài AI, nhóm tham khảo cấu trúc mã
khung của một bản cài đặt tham khảo và dùng nó sinh dữ liệu kiểm thử chuẩn cho từng tầng;
báo cáo dùng mẫu hcmut-report \cite{hcmutreport}.
