\subsection{Nguồn dữ liệu và mã tham khảo}

\begin{itemize}
  \item \textbf{Đề bài, KB và scaffolding}: do giảng viên cung cấp \cite{spec}.
  \item \textbf{Dữ liệu nhóm viết}: 22 câu thử thách và 32 cách hỏi trong câu đầu vào, chỉ
        dùng để kiểm thử.
  \item \textbf{Bản cài đặt tham khảo}: nhóm tham khảo cấu trúc mã khung và định dạng
        output, dùng nó sinh dữ liệu kiểm thử chuẩn cho từng tầng. Mã nộp do nhóm tự viết,
        kiểm thử và giải thích được.
  \item \textbf{Lý thuyết}: thuật toán Earley \cite{earley1970}; CFG và văn phạm ngữ nghĩa
        theo giáo trình \cite{phanthituoi,jurafsky,bird}.
  \item \textbf{Mẫu báo cáo}: hcmut-report \cite{hcmutreport}, giấy phép LPPL 1.3c.
\end{itemize}

\subsection{Khai báo sử dụng AI}

Theo quy định sử dụng AI của môn, nhóm khai báo việc dùng AI ở \cref{tab:ai}.
\textbf{Khi chạy, hệ thống không dùng LLM hay mô hình học máy nào}; AI chỉ hỗ trợ trong quá
trình phát triển.

\begin{table}[htbp]
  \centering
  \small
  \begin{tabularx}{\linewidth}{|P{3cm}|L|}
    \hline
    \textbf{Công cụ} & \textbf{Phạm vi hỗ trợ} \\ \hline
    Claude Code: Claude Opus, Claude Sonnet; 22/10--19/11/2026 &
      Gộp grammar giai đoạn 1; hợp đồng giữa các tầng và công cụ sinh golden test;
      pipeline, CLI, xuất output, đánh giá; sửa tokenizer và bộ sinh câu; kiểm chứng
      Docker; soạn và biên tập báo cáo \LaTeX. Commit liên quan có dòng
      \texttt{Co-Authored-By: Claude}. \\ \hline
    Codex; 16--17/11/2026 &
      Sửa luật rỗng và kiểm tra mơ hồ của parser; mở rộng grammar và semantic cho tính
      năng BTL; câu thử thách C009--C022; công cụ kiểm tra grammar và đóng gói; nháp
      README, báo cáo, kịch bản demo; bổ sung test phản ví dụ. \\ \hline
  \end{tabularx}
  \caption{Khai báo sử dụng công cụ AI}
  \label{tab:ai}
\end{table}

Mọi thay đổi có AI hỗ trợ đều phải qua test hợp đồng, test hồi quy và parse lại toàn bộ câu
sinh, rồi được thành viên review trước khi đưa vào bản nộp. Nhóm không sửa KB để chữa ca
sai và chịu trách nhiệm giải thích được toàn bộ mã nộp.
